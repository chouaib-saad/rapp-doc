  % Les macros d'échelle rendent une expression NUE : chaque
  % emploi la place lui-même entre accolades, ce qui autorise
  % une arithmétique supplémentaire sans imbriquer les groupes.
  \def\Y#1{(#1-0.75)*36}
  \def\Z#1{#1*6.1}

  % ══════════════════════════════════════════════════════════

    \node[font=\sffamily\tiny\itshape, anchor=south, rotate=90] at (-1.05,3.2)
      {score de proximité};

      {Le bruit hors domaine dépasse la réponse correcte en arabe. Aucun seuil ne sépare les deux distributions.};
